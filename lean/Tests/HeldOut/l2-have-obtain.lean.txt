-- id: l2-have-obtain
-- level: 2
-- kind: alternative
-- note: the demo's `have`, then destructured
import Distill.Levels.Demo
open Distill Distill.Levels.Demo
namespace Catalog

theorem inter_subset_left (A B : Set α) : A ∩ B ⊆ A := by
  intro x hx
  have h : x ∈ A ∧ x ∈ B := hx
  obtain ⟨ha, hb⟩ := h
  exact ha
