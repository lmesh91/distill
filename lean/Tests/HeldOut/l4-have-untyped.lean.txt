-- id: l4-have-untyped
-- level: 4
-- kind: alternative
-- note: `have` without types
import Distill.Levels.Demo
open Distill Distill.Levels.Demo
namespace Catalog

theorem subset_inter {A B C : Set α} (hCA : C ⊆ A) (hCB : C ⊆ B) : C ⊆ A ∩ B := by
  intro x hx
  have ha := hCA x hx
  have hb := hCB x hx
  exact ⟨ha, hb⟩
