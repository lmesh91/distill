-- id: l4-have-by
-- level: 4
-- kind: alternative
-- note: `have`s proved by nested tactic blocks
import Distill.Levels.Demo
open Distill Distill.Levels.Demo
namespace Catalog

theorem subset_inter {A B C : Set α} (hCA : C ⊆ A) (hCB : C ⊆ B) : C ⊆ A ∩ B := by
  intro x hx
  have ha : x ∈ A := by
    exact hCA x hx
  have hb : x ∈ B := by
    exact hCB x hx
  exact ⟨ha, hb⟩
