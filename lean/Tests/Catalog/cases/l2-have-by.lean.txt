-- id: l2-have-by
-- level: 2
-- kind: alternative
-- note: `have` proved by a nested tactic block
import Distill.Levels.Demo
open Distill Distill.Levels.Demo
namespace Catalog

theorem inter_subset_left (A B : Set α) : A ∩ B ⊆ A := by
  intro x hx
  have h : x ∈ A ∧ x ∈ B := by
    exact hx
  exact h.1
