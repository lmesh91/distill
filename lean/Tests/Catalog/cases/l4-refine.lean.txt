-- id: l4-refine
-- level: 4
-- kind: alternative
-- note: `refine ⟨?_, ?_⟩` with bullets
import Distill.Levels.Demo
open Distill Distill.Levels.Demo
namespace Catalog

theorem subset_inter {A B C : Set α} (hCA : C ⊆ A) (hCB : C ⊆ B) : C ⊆ A ∩ B := by
  intro x hx
  refine ⟨?_, ?_⟩
  · exact hCA x hx
  · exact hCB x hx
