theorem inter_subset_inter_comm (A B : Set α) : A ∩ B ⊆ B ∩ A := by
  have hA : A ∩ B ⊆ A := inter_subset_left A B
  have hB : A ∩ B ⊆ B := inter_subset_right A B
  exact subset_inter hA hB
