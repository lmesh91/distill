-- id: l5-have-swapped
-- level: 5
-- kind: alternative
-- note: the `have`s in the other order, untyped
import Distill.Levels.Demo
open Distill Distill.Levels.Demo
namespace Catalog

theorem inter_subset_inter_comm (A B : Set α) : A ∩ B ⊆ B ∩ A := by
  have hB := inter_subset_right A B
  have hA := inter_subset_left A B
  exact subset_inter hB hA
