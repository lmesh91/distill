-- level: 4
-- kind: alternative
-- note: a hypothesis applied to the element only
import Distill.Levels.Demo
open Distill Distill.Levels.Demo
namespace Catalog

theorem subset_inter {A B C : Set α} (hCA : C ⊆ A) (hCB : C ⊆ B) : C ⊆ A ∩ B := by
  intro x hx
  have hA := hCA x
  exact ⟨hA hx, hCB x hx⟩
